        async function submitFeatures() {
            const username = document.getElementById('username').value;
            const fullname = document.getElementById('fullname').value;
            const profilePic = parseInt(document.getElementById('profilePic').value);
            const privateAccount = parseInt(document.getElementById('private').value);
            const posts = parseInt(document.getElementById('posts').value);
            const followers = parseInt(document.getElementById('followers').value);
            const follows = parseInt(document.getElementById('follows').value);

            // Calculate dynamic features
            const numsLengthUsername = username ? username.replace(/\D/g, '').length / username.length : 0;
            const fullnameWords = fullname ? fullname.split(' ').length : 0;
            const numsLengthFullname = fullname ? fullname.replace(/\D/g, '').length / fullname.length : 0;
            const nameEqualsUsername = fullname === username ? 1 : 0;

            const features = {
                username,
                'profile pic': profilePic,
                'nums/length username': numsLengthUsername,
                'fullname words': fullnameWords,
                'nums/length fullname': numsLengthFullname,
                'name==username': nameEqualsUsername,
                'description length': 80, // Mock value
                'external URL': 1, // Mock value
                'private': privateAccount,
                '#posts': posts,
                '#followers': followers,
                '#follows': follows
            };

            // Send features to the predict route
            try {
                const response = await fetch('/predict_features', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(features)
                });

                const result = await response.json();
                if (response.ok) {
                    document.getElementById('result').innerHTML = `
                        <div class="alert alert-success" role="alert">
                            Prediction: <strong>${result.prediction}</strong>
                        </div>`;
                } else {
                    document.getElementById('result').innerHTML = `
                        <div class="alert alert-danger" role="alert">
                            Error: ${result.error}
                        </div>`;
                }
            } catch (error) {
                document.getElementById('result').innerHTML = `
                    <div class="alert alert-danger" role="alert">
                        Error: ${error.message}
                    </div>`;
            }
        }

        async function sendQuery() {
            const query = document.getElementById("query").value;

            if (!query) {
                alert("Please write your query.");
                return;
            }

            try {
                const response = await fetch("/send_query", {
                    method: "POST",
                    headers: {
                        "Content-Type": "application/json"
                    },
                    body: JSON.stringify({ query })
                });

                const result = await response.json();

                if (response.ok) {
                    alert("Your query has been sent successfully!");
                    document.getElementById("contactForm").reset();
                } else {
                    alert(`Error: ${result.error}`);
                }
            } catch (error) {
                alert(`Error: ${error.message}`);
            }
        }
    </script>
